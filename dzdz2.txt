/*задание2*/
#include <iostream>
#include <fstream>
using namespace std;

class Tochka
{
private:
    int x;
    int y;
    int z;

public:
    void vvod()
    {
        cout << "введите x: ";
        cin >> x;

        cout << "введите y: ";
        cin >> y;

        cout << "введите z: ";
        cin >> z;
    }

    void pokaz()
    {
        cout << "x = " << x << endl;
        cout << "y = " << y << endl;
        cout << "z = " << z << endl;
    }

    int getX()
    {
        return x;
    }

    int getY()
    {
        return y;
    }

    int getZ()
    {
        return z;
    }

    void sohranit()
    {
        ofstream file("tochka.txt");

        file << x << endl;
        file << y << endl;
        file << z << endl;

        file.close();
    }

    void zagruzit()
    {
        ifstream file("tochka.txt");

        file >> x;
        file >> y;
        file >> z;

        file.close();
    }
};

int main()
{
    Tochka a;

    a.vvod();

    cout << endl;
    a.pokaz();

    cout << endl;

    cout << "X через акцесор: " << a.getX() << endl;
    cout << "Y через акцесор: " << a.getY() << endl;
    cout << "Z через акцесор: " << a.getZ() << endl;

    a.sohranit();

    cout << endl;
    cout << "данные сохранены в файл." << endl;

    a.zagruzit();

    cout << "данные загружены с файла:" << endl;
    a.pokaz();

    return 0;
}
